=== order 1, M=1 ===
∂ₜ(⟨a⟩) = η - ⟨σ₁₂⟩*g1*im + ⟨a⟩*(-(1//2)*κt - im*Δ0)
∂ₜ(⟨σ₂₁⟩) = ⟨a'⟩*g1*im + ⟨σ₂₁⟩*im*δ1 - 2⟨a'⟩*⟨σ₂₂⟩*g1*im
∂ₜ(⟨σ₂₂⟩) = ⟨a'⟩*⟨σ₁₂⟩*g1*im - ⟨σ₂₁⟩*⟨a⟩*g1*im


=== order 2, M=1 ===
∂ₜ(⟨a⟩) = η - ⟨σ₁₂⟩*g1*im + ⟨a⟩*(-(1//2)*κt - im*Δ0)
∂ₜ(⟨σ₂₁⟩) = ⟨a'⟩*g1*im - 2⟨a' * σ₂₂⟩*g1*im + ⟨σ₂₁⟩*im*δ1
∂ₜ(⟨σ₂₂⟩) = ⟨a' * σ₁₂⟩*g1*im - ⟨a * σ₂₁⟩*g1*im
∂ₜ(⟨a' * σ₂₂⟩) = ⟨σ₂₂⟩*η + ⟨a' * σ₂₂⟩*(-(1//2)*κt + im*Δ0) + (⟨σ₁₂⟩*⟨a' * a'⟩ + 2⟨a'⟩*⟨a' * σ₁₂⟩ - 2⟨σ₁₂⟩*(⟨a'⟩^2))*g1*im - (⟨a⟩*⟨a' * σ₂₁⟩ + ⟨a'⟩*⟨a * σ₂₁⟩ + ⟨a' * a⟩*⟨σ₂₁⟩ - 2⟨a'⟩*⟨σ₂₁⟩*⟨a⟩)*g1*im
∂ₜ(⟨a * σ₂₁⟩) = ⟨σ₂₁⟩*η + ⟨a' * a⟩*g1*im - ⟨σ₂₂⟩*g1*im + ⟨a * σ₂₁⟩*(-(1//2)*κt + im*(-Δ0 + δ1)) - 2(⟨a⟩*⟨a' * σ₂₂⟩ + ⟨a'⟩*⟨a * σ₂₂⟩ + ⟨a' * a⟩*⟨σ₂₂⟩ - 2⟨a'⟩*⟨σ₂₂⟩*⟨a⟩)*g1*im
∂ₜ(⟨a' * a⟩) = ⟨a⟩*η + ⟨a'⟩*η - ⟨a' * a⟩*κt + ⟨a * σ₂₁⟩*g1*im - ⟨a' * σ₁₂⟩*g1*im
∂ₜ(⟨a' * σ₂₁⟩) = ⟨σ₂₁⟩*η + ⟨a' * a'⟩*g1*im + ⟨a' * σ₂₁⟩*(-(1//2)*κt + im*(Δ0 + δ1)) - 2(⟨σ₂₂⟩*⟨a' * a'⟩ + 2⟨a'⟩*⟨a' * σ₂₂⟩ - 2⟨σ₂₂⟩*(⟨a'⟩^2))*g1*im
∂ₜ(⟨a' * a'⟩) = 2⟨a'⟩*η + 2⟨a' * σ₂₁⟩*g1*im + ⟨a' * a'⟩*(-κt + 2im*Δ0)
